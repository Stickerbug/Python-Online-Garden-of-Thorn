18194     def _atomic_arctic_nuke(self, player_id, card, params, log, choice, context):
18195         target_id = self._resolve_target(player_id, params.get('target', 'target'))
18196         if not self._valid_player_id(target_id):
18197             return
18198         player = self.players[player_id]
18199         # Normal cost modifiers (same-name cost, Frost, temporary Heavy, etc.)
18200         # are paid before on_play runs.  They are still part of the Nuke's
18201         # "consume all E", so include them instead of producing zero hits when
18202         # they happened to consume the player's last E.
18203         paid_elixir = max(0, int(getattr(card, '_paid_e_this_play', 0) or 0))
18204         remaining_elixir = max(0, int(getattr(player, 'elixir', 0) or 0))
18205         spent_elixir = paid_elixir + remaining_elixir
18206         if remaining_elixir > 0:
18207             self._spend_resource(player_id, 'elixir', remaining_elixir, card)
18208         if spent_elixir <= 0:
18209             return
18210         repeats = max(
18211             1,
18212             clamp_card_layer(getattr(card, 'fission_level', 1))
18213             + clamp_card_extra_hits(getattr(card, 'extra_hits', 0)),
18214         )
18215         percent = max(0.0, float(params.get('health_percent', 0.08) or 0.08))
18216         minimum = max(0, self._eval_int(player_id, params.get('minimum', 2), card, 2))
18217         self._game_over_defer_depth += 1
18218         first_attack = True
18219         try:
18220             for _ in range(spent_elixir):
18221                 for _ in range(repeats):
18222                     if self.players[target_id].health <= 0:
18223                         break
18224                     base_amount = max(minimum, int(math.ceil(self.players[target_id].health * percent)))
18225                     amount = self._modified_attack_damage(base_amount, card)
18226                     self.deal_attack_damage(
18227                         target_id,
18228                         amount,
18229                         1,
18230                         attacker_id=player_id,
18231                         source_card=card,
18232                     )
18233                     if first_attack:
18234                         card.power_value = 0
18235                         card.instance_flags.discard('power')
18236                         first_attack = False
18237         finally:
18238             self._game_over_defer_depth -= 1
18239         card.fission_level = 1
18240         card.fission_count = 0
18241         card.extra_hits = 0
18242         self._check_game_over()
